import pandas as pd
from sklearn.model_selection import train_test_split

# Load dataset
df = pd.read_csv("Social_Network_Ads")

# Features & target
X = df[['Age', 'EstimatedSalary']]
y = df['Purchased']

# Train-test split
X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.25, random_state=42
)
